cmake_minimum_required(VERSION 3.24)
project(TheSoleSurvivorDreams VERSION 0.1.0 LANGUAGES CXX)
option(DREAMS_TESTS_ONLY "Build portable logic tests without Windows/F4SE" OFF)
if(DREAMS_TESTS_ONLY)
  enable_testing()
  add_executable(dreams_tests tests/DreamsTests.cpp)
  target_include_directories(dreams_tests PRIVATE src)
  target_compile_features(dreams_tests PRIVATE cxx_std_20)
  add_test(NAME dreams_logic COMMAND dreams_tests)
  return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The plugin requires Windows x64 and Visual Studio 2022. Use DREAMS_TESTS_ONLY on Linux.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(Boost_USE_STATIC_LIBS ON)
set(Boost_USE_STATIC_RUNTIME OFF)
include(FetchContent)
FetchContent_Declare(CommonLibF4RD
  GIT_REPOSITORY https://github.com/Zzyxz/CommonLibF4RD.git
  GIT_TAG b092bcc70f690b2c51b980c0b251fa56bf2d3460)
set(COMMONLIBF4RD_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(CommonLibF4RD)
add_library(TheSoleSurvivorDreams SHARED src/Plugin.cpp)
target_compile_features(TheSoleSurvivorDreams PRIVATE cxx_std_20)
target_compile_definitions(TheSoleSurvivorDreams PRIVATE _UNICODE)
target_compile_options(TheSoleSurvivorDreams PRIVATE /utf-8 /permissive- /Zc:preprocessor /EHsc)
target_precompile_headers(TheSoleSurvivorDreams PRIVATE src/PCH.h)
target_link_libraries(TheSoleSurvivorDreams PRIVATE CommonLibF4::CommonLibF4)
install(TARGETS TheSoleSurvivorDreams RUNTIME DESTINATION F4SE/Plugins)
